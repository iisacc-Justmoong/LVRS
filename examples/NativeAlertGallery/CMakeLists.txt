cmake_minimum_required(VERSION 3.31)
project(LVRSNativeAlertGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME Gallery SOURCE Gallery.lvui OUTPUT_VARIABLE view FRAMEWORK LVRS INCLUDES GalleryBackend.h)
iisdkcore_add_application(LVRSNativeAlertGallery GUI SOURCES main.cpp "${view}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeAlertGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
if(APPLE)
    set_target_properties(LVRSNativeAlertGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativealertgallery" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native Alert Gallery")
endif()
if(MSVC)
    target_compile_options(LVRSNativeAlertGallery PRIVATE /W4 /WX)
else()
    target_compile_options(LVRSNativeAlertGallery PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    add_test(NAME LVRS.native_alert_gallery COMMAND LVRSNativeAlertGallery --self-test)
    set_tests_properties(LVRS.native_alert_gallery PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endif()
