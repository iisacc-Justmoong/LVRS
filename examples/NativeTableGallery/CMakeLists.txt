cmake_minimum_required(VERSION 3.31)
project(LVRSNativeTableGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME Gallery SOURCE Gallery.lvui OUTPUT_VARIABLE view FRAMEWORK LVRS INCLUDES GalleryBackend.h)
iisdkcore_add_application(LVRSNativeTableGallery GUI SOURCES main.cpp "${view}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeTableGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
if(APPLE)
    set_target_properties(LVRSNativeTableGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativetablegallery" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native Table Gallery")
endif()
if(BUILD_TESTING)
    add_test(NAME LVRS.native_table_gallery COMMAND LVRSNativeTableGallery --self-test)
    set_tests_properties(LVRS.native_table_gallery PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endif()
