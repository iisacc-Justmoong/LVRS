cmake_minimum_required(VERSION 3.31)
project(LVRSNativeDataTableGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME Gallery SOURCE Gallery.lvui OUTPUT_VARIABLE view FRAMEWORK LVRS INCLUDES GalleryBackend.h)
iisdkcore_add_application(LVRSNativeDataTableGallery GUI SOURCES main.cpp "${view}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeDataTableGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
if(APPLE)
    set_target_properties(LVRSNativeDataTableGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativedatatablegallery" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native Data Table Gallery")
endif()
if(BUILD_TESTING)
    add_test(NAME LVRS.native_data_table_gallery COMMAND LVRSNativeDataTableGallery --self-test)
    set_tests_properties(LVRS.native_data_table_gallery PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endif()
