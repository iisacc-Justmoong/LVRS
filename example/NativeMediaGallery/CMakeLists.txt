cmake_minimum_required(VERSION 3.31)
project(LVRSNativeMediaGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME Media SOURCE App/Gallery.lvui OUTPUT_VARIABLE gallery FRAMEWORK LVRS INCLUDES Backend.h)
iisdkcore_add_application(LVRSNativeMediaGallery GUI SOURCES App/main.cpp App/Backend.cpp "${gallery}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeMediaGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/App")
set_target_properties(LVRSNativeMediaGallery PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(LVRSNativeMediaGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativemediagallery" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native Media Gallery")
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS LVRSNativeMediaGallery RUNTIME DESTINATION bin BUNDLE DESTINATION .)
endif()
