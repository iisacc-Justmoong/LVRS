cmake_minimum_required(VERSION 3.31)
project(LVRSNativeCreationWorkbench LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME Workbench SOURCE App/Workbench.lvui OUTPUT_VARIABLE workbench FRAMEWORK LVRS INCLUDES Backend.h)
iisdkcore_add_application(LVRSNativeCreationWorkbench GUI SOURCES App/main.cpp App/Backend.cpp "${workbench}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeCreationWorkbench PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/App")
set_target_properties(LVRSNativeCreationWorkbench PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(LVRSNativeCreationWorkbench PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativecreationworkbench" MACOSX_BUNDLE_BUNDLE_NAME "Native Creation Workbench")
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS LVRSNativeCreationWorkbench RUNTIME DESTINATION bin BUNDLE DESTINATION .)
endif()
