cmake_minimum_required(VERSION 3.31)
project(LVRSNativeTooltipGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME TooltipGallery SOURCE App/Gallery.lvui OUTPUT_VARIABLE gallery FRAMEWORK LVRS INCLUDES Backend.h)
iisdkcore_add_application(LVRSNativeTooltipGallery GUI SOURCES App/main.cpp App/Backend.cpp "${gallery}"
    LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeTooltipGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/App")
set_target_properties(LVRSNativeTooltipGallery PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(LVRSNativeTooltipGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativetooltip" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native Tooltip")
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS LVRSNativeTooltipGallery RUNTIME DESTINATION bin BUNDLE DESTINATION .)
endif()
