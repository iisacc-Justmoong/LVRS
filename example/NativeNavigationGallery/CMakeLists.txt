cmake_minimum_required(VERSION 3.31)
project(LVRSNativeNavigationGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME NavigationGallery SOURCE App/Gallery.lvui OUTPUT_VARIABLE view FRAMEWORK LVRS INCLUDES Backend.h)
iisdkcore_add_application(LVRSNativeNavigationGallery GUI SOURCES App/main.cpp App/Backend.cpp "${view}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeNavigationGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/App")
set_target_properties(LVRSNativeNavigationGallery PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(LVRSNativeNavigationGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativenavigationgallery" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native Navigation Model")
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS LVRSNativeNavigationGallery RUNTIME DESTINATION bin BUNDLE DESTINATION .)
endif()
