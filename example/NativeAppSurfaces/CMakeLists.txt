cmake_minimum_required(VERSION 3.31)
project(LVRSNativeAppSurfaces LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME Surfaces SOURCE App/Gallery.lvui OUTPUT_VARIABLE gallery FRAMEWORK LVRS INCLUDES Backend.h)
iisdkcore_add_application(LVRSNativeAppSurfaces GUI SOURCES App/main.cpp App/Backend.cpp "${gallery}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeAppSurfaces PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/App")
set_target_properties(LVRSNativeAppSurfaces PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(LVRSNativeAppSurfaces PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativeappsurfaces" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native App Surfaces")
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS LVRSNativeAppSurfaces RUNTIME DESTINATION bin BUNDLE DESTINATION .)
endif()
