cmake_minimum_required(VERSION 3.31)
project(LVRSNativePageRouterGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME RouterGallery SOURCE App/Gallery.lvui OUTPUT_VARIABLE gallery FRAMEWORK LVRS INCLUDES Backend.h)
lvui_generate(NAME HomePage SOURCE App/Home.lvui OUTPUT_VARIABLE home FRAMEWORK LVRS INCLUDES PageBackend.h)
lvui_generate(NAME ReportPage SOURCE App/Report.lvui OUTPUT_VARIABLE report FRAMEWORK LVRS INCLUDES PageBackend.h)
iisdkcore_add_application(LVRSNativePageRouterGallery GUI SOURCES App/main.cpp App/Backend.cpp App/PageBackend.cpp "${gallery}" "${home}" "${report}" LIBRARIES LVRS::LVRS RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativePageRouterGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/App")
set_target_properties(LVRSNativePageRouterGallery PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(LVRSNativePageRouterGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativepageroutergallery" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native Pages")
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS LVRSNativePageRouterGallery RUNTIME DESTINATION bin BUNDLE DESTINATION .)
endif()
