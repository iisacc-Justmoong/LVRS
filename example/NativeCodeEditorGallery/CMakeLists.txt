cmake_minimum_required(VERSION 3.31)
project(LVRSNativeCodeEditorGallery LANGUAGES CXX)
include(CTest)
if(NOT TARGET LVRS::LVRS)
    find_package(LVRS 1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME CodeEditorGallery SOURCE App/Gallery.lvui OUTPUT_VARIABLE gallery FRAMEWORK LVRS INCLUDES Backend.h)
iisdkcore_add_application(LVRSNativeCodeEditorGallery GUI SOURCES App/main.cpp App/Backend.cpp "${gallery}"
    LIBRARIES LVRS::LVRS iiFileProvider::Native RESOURCES ${LVRS_NATIVE_RESOURCES})
target_include_directories(LVRSNativeCodeEditorGallery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/App")
set_target_properties(LVRSNativeCodeEditorGallery PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
    set_target_properties(LVRSNativeCodeEditorGallery PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.lvrs.nativecodeeditor" MACOSX_BUNDLE_BUNDLE_NAME "LVRS Native CodeEditor")
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS LVRSNativeCodeEditorGallery RUNTIME DESTINATION bin BUNDLE DESTINATION .)
endif()
