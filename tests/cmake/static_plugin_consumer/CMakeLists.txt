cmake_minimum_required(VERSION 3.21)
project(LVRSStaticPluginConsumerContract LANGUAGES CXX)

if(LVRS_PLUGIN_SCENARIO STREQUAL "package")
    find_package(LVRS CONFIG REQUIRED)
    get_target_property(package_links LVRS::LVRS INTERFACE_LINK_LIBRARIES)
    if(NOT "iiFileProvider::iiFileProvider" IN_LIST package_links)
        message(FATAL_ERROR "The installed static LVRS target must propagate iiFileProvider")
    endif()
else()
    include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/LVRSHelpers.cmake")
endif()

if(LVRS_PLUGIN_SCENARIO STREQUAL "installed")
    if(NOT EXISTS "${LVRS_PLUGIN_ARCHIVE}")
        message(FATAL_ERROR "The installed static plugin archive is required")
    endif()
    add_library(LVRS::LVRSCoreplugin UNKNOWN IMPORTED)
    set_target_properties(LVRS::LVRSCoreplugin PROPERTIES IMPORTED_LOCATION "${LVRS_PLUGIN_ARCHIVE}")
    add_library(LVRSCoreplugin INTERFACE IMPORTED)
    set_target_properties(LVRSCoreplugin PROPERTIES INTERFACE_LINK_LIBRARIES LVRS::LVRSCoreplugin)
    set(expected_target LVRS::LVRSCoreplugin)
elseif(LVRS_PLUGIN_SCENARIO STREQUAL "build-tree")
    add_library(LVRSCoreplugin STATIC plugin.cpp)
    set(expected_target LVRSCoreplugin)
elseif(LVRS_PLUGIN_SCENARIO STREQUAL "package")
    if(NOT TARGET LVRS::LVRSCoreplugin)
        message(FATAL_ERROR "The installed package is missing its static QML plugin")
    endif()
    get_target_property(package_plugin LVRS::LVRSCoreplugin IMPORTED_LOCATION)
    if(NOT EXISTS "${package_plugin}")
        message(FATAL_ERROR "The installed static QML plugin archive is missing")
    endif()
    set(expected_target LVRS::LVRSCoreplugin)
else()
    message(FATAL_ERROR "Select LVRS_PLUGIN_SCENARIO=installed, build-tree, or package")
endif()

if(LVRS_PLUGIN_SCENARIO STREQUAL "package")
    qt_add_executable(consumer MANUAL_FINALIZATION main.cpp)
    target_link_libraries(consumer PRIVATE LVRS::LVRS)
else()
    add_executable(consumer main.cpp)
endif()
_lvrs_internal_maybe_link_static_lvrs_plugin(consumer)
_lvrs_internal_maybe_link_static_lvrs_plugin(consumer)
get_target_property(linked consumer LINK_LIBRARIES)
if(NOT expected_target IN_LIST linked)
    message(FATAL_ERROR "The consumer did not link the actual static plugin: ${linked}")
endif()
get_target_property(sources consumer SOURCES)
set(import_count 0)
foreach(source IN LISTS sources)
    if(source MATCHES "consumer_lvrs_plugin_import[.]cpp$")
        math(EXPR import_count "${import_count} + 1")
        file(READ "${source}" import_source)
        if(NOT import_source MATCHES "Q_IMPORT_PLUGIN\\(LVRSPlugin\\)")
            message(FATAL_ERROR "The consumer import does not name LVRSPlugin")
        endif()
    endif()
endforeach()
if(NOT import_count EQUAL 1)
    message(FATAL_ERROR "The static plugin import must be added exactly once")
endif()
if(LVRS_PLUGIN_SCENARIO STREQUAL "package")
    qt_finalize_executable(consumer)
endif()
