cmake_minimum_required(VERSION 3.21)
project(LVRSInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(LVRS_CONSUMER_REQUIRE_IMPLICIT_LIBRARY_PATH
    "Verify Darwin RPATH propagation when LIBRARY_PATH makes LVRS implicit" OFF)
find_package(LVRS CONFIG REQUIRED)
get_target_property(_lvrs_library LVRS::LVRS IMPORTED_LOCATION_RELEASE)
if(NOT _lvrs_library)
    get_target_property(_lvrs_library LVRS::LVRS IMPORTED_LOCATION)
endif()
get_filename_component(_lvrs_library_directory "${_lvrs_library}" DIRECTORY)
if(LVRS_CONSUMER_REQUIRE_IMPLICIT_LIBRARY_PATH)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        message(FATAL_ERROR "The implicit-library-path regression requires Darwin.")
    endif()
    list(FIND CMAKE_CXX_IMPLICIT_LINK_DIRECTORIES
        "${_lvrs_library_directory}" _lvrs_implicit_index)
    if(_lvrs_implicit_index EQUAL -1)
        message(FATAL_ERROR "Set LIBRARY_PATH to the installed LVRS library directory before a fresh configure.")
    endif()
    get_target_property(_lvrs_link_options LVRS::LVRS INTERFACE_LINK_OPTIONS)
    if(NOT "LINKER:-rpath,${_lvrs_library_directory}" IN_LIST _lvrs_link_options)
        message(FATAL_ERROR "LVRS must preserve its own RPATH when its library directory is implicit.")
    endif()
endif()
add_executable(lvrs_installed_consumer main.cpp)
target_link_libraries(lvrs_installed_consumer PRIVATE LVRS::LVRS)
enable_testing()
add_test(NAME installed_lvrs_qml_consumer COMMAND lvrs_installed_consumer)
set_tests_properties(installed_lvrs_qml_consumer PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QML2_IMPORT_PATH=${LVRS_QML_IMPORT_PATH}")
